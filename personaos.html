Quick Start — eafit-challenge-agent-example
# 1. Clonar el fork (ya lo hiciste)
$ git clone https://github.com/Danielsma803/eafit-challenge-agent-example.git
$ cd eafit-challenge-agent-example

# 2. Pegar archivos generados en Step 4 del wizard de PersonaOS
$ cp ~/Downloads/agent-pack.yaml app/agent-packs/example-agent/agent-pack.yaml
$ cp ~/Downloads/config.env .

# 3. Setup: conecta con org EAFIT y obtiene Service Credential
$ source config.env && ./scripts/setup.sh
# → VS Agent Admin: http://localhost:3010
# → VS Agent Public: http://localhost:3011 (exponer con ngrok)

# 4. Iniciar stack Docker (chatbot:3003 + redis + postgres)
$ export OPENAI_API_KEY=sk-... # o ANTHROPIC_API_KEY
$ ./scripts/start.sh

# 5. Exponer puerto 3011 para Hologram (DIDComm público)
$ ngrok http 3011
# → Copiar URL ngrok → actualizar AGENT_PUBLIC_URL en config.env

# 6. Verificar servicios
$ curl http://localhost:3010/agent-info # VS Agent admin
$ curl http://localhost:3003/health # Chatbot
$ curl http://localhost:3011 # VS Agent público (DIDComm)
